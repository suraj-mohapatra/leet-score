Be careful: this solution works, but it exceeds the LeetCode time limit.

The outer loop and the two-pointer loop have a combined complexity of O(n^2).
Sorting the array with Java's `Arrays.sort` takes O(n log n), so the overall loop
complexity is O(n^2).

The `result.contains(triplet)` check is the likely bottleneck because it repeatedly
searches the result list. It can be replaced with duplicate-skipping logic on
`nums[p1]` and `nums[p2]` to avoid this extra search.
  ```
class Solution {
    public List<List<Integer>> threeSum(int[] nums) {

        List<List<Integer>> result = new ArrayList<>();

        Arrays.sort(nums);

        for (int i = 0; i < nums.length; i++) {

            if (i > 0 && nums[i] == nums[i - 1]) {
                continue;
            }
            int target = -nums[i];

            int p1 = i + 1;
            int p2 = nums.length - 1;

            while (p1 < p2) {

                if (nums[p1] + nums[p2] < target) {
                    p1++;
                } else if (nums[p1] + nums[p2] > target) {
                    p2--;
                } else {
                    List<Integer> triplet = List.of(nums[i], nums[p1], nums[p2]);
                    if (!result.contains(triplet)) {

                        result.add(triplet);

                    }
                    //result.contains(triplet) ? result.add(null) : ;
                    p1++;
                    p2--;
                }

            }
        }

        return result;

    }
}
```

```
class Solution {
    public List<List<Integer>> threeSum(int[] nums) {

        List<List<Integer>> result = new ArrayList<>();

        Arrays.sort(nums);

        for (int i = 0; i < nums.length; i++) {

            if (i > 0 && nums[i] == nums[i - 1]) {
                continue;
            }
            int target = -nums[i];

            int p1 = i + 1;
            int p2 = nums.length - 1;

            while (p1 < p2) {

                if (nums[p1] + nums[p2] < target) {
                    p1++;
                } else if (nums[p1] + nums[p2] > target) {
                    p2--;
                } else {
                    List<Integer> triplet = List.of(nums[i], nums[p1], nums[p2]);

                    result.add(triplet);

                    //result.contains(triplet) ? result.add(null) : ;
                    p1++;
                    p2--;

                    while (nums[p1] == nums[p1 - 1] && p1 < p2 ) {
                        p1++;
                    }

                    while (nums[p2] == nums[p2 + 1] && p1 < p2) {
                        p2--;
                    }
                }

            }
        }

        return result;

    }
}
```
